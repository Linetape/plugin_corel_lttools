Private Sub CarregarIcone(ByVal i As Long, ByVal x As Single, ByVal y As Single)
    Dim bmp As String, im As MSForms.Image, arq As String
    arq = gMenuIco(i)
    bmp = Left$(arq, Len(arq) - 4) & ".bmp"      ' BMP é o formato mais seguro para o VBA
    If Len(Dir$(bmp)) = 0 Then bmp = arq

    With im

        .BorderStyle = fmBorderStyleNone
        .BackStyle = fmBackStyleTransparent
        .PictureSizeMode = fmPictureSizeModeZoom
        On Error Resume Next
        Set .Picture = LoadPicture(bmp)
    End With
End Sub

Private Sub PosicionarNoMouse()
    Dim p As PONTO, dpi As Long
    #If VBA7 Then
        Dim dc As LongPtr
    #Else
        Dim dc As Long
    #End If
    On Error Resume Next
    GetCursorPos p
    dc = GetDC(0)
    dpi = GetDeviceCaps(dc, 88)
    ReleaseDC 0, dc
    If dpi <= 0 Then dpi = 96
    Me.StartUpPosition = 0
    Me.Left = p.X * 72 / dpi - 10
    Me.Top = p.Y * 72 / dpi + 8
End Sub

Private Function Numero(ByVal s As String, ByRef v As Double) As Boolean
    Dim sep As String
    sep = Mid$(CStr(1.5), 2, 1)
    s = Replace(Replace(Trim$(s), ",", sep), ".", sep)
    If Len(s) = 0 Or Not IsNumeric(s) Then Exit Function
    v = CDbl(s)
    Numero = True
End Function

Private Function PassoDe(ByVal s As String) As Double
    Dim v As Double
    If Not Numero(s, v) Then PassoDe = 1: Exit Function
    v = Abs(v)
    If v = 0 Or v >= 1 And v = Int(v) Then
        PassoDe = 1
    ElseIf v < 0.1 Then
        PassoDe = 0.01
    Else
        PassoDe = 0.1
    End If
End Function

Private Sub Girar(sp As MSForms.SpinButton, ByVal dirc As Double)
    Dim i As Long, v As Double
    i = CLng(sp.Tag)
    If Not Numero(ctl(i).Text, v) Then v = 0
    v = Round(v + dirc * passo(i), 4)
    If v < 0 Then v = 0
    ctl(i).Text = CStr(v)
End Sub

' lê e confere os campos (números válidos) e guarda em gVal
Private Function LerCampos() As Boolean
    Dim i As Long, v As Double
    For i = 1 To gN
        Select Case gTipo(i)
            Case 1
                If Not Numero(ctl(i).Text, v) Then
                    MsgBox "Valor inválido em: " & gRot(i), vbExclamation, "LT Tools"
                    ctl(i).SetFocus
                    Exit Function
                End If
                gVal(i) = Trim$(ctl(i).Text)
            Case 2
                gVal(i) = IIf(ctl(i).Value, "1", "0")
            Case 3
                gVal(i) = CStr(ctl(i).ListIndex + 1)
            Case 4
                gVal(i) = ctl(i).Text
        End Select
    Next
    LerCampos = True
End Function

Private Function Chave() As String
    Chave = "Pre_" & gTit
End Function

Private Sub PreencherLista(ByVal selecionar As String)
    Dim v As Variant, i As Long
    carregando = True
    cbPre.Clear
    cbPre.AddItem "(último usado)"
    cbPre.AddItem "(valores de fábrica)"
    If Len(GetSetting("LTTools", Chave(), "_padrao", "")) > 0 Then cbPre.AddItem "(padrão)"
    For Each v In Split(GetSetting("LTTools", Chave(), "_lista", ""), "|")
        If Len(v) > 0 Then cbPre.AddItem v
    Next
    If Len(selecionar) > 0 Then
        For i = 0 To cbPre.ListCount - 1
            If cbPre.List(i) = selecionar Then cbPre.ListIndex = i: Exit For
        Next
    End If
    carregando = False
End Sub

Private Sub cbPre_Change()
    If carregando Or cbPre.ListIndex < 0 Then Exit Sub
    Select Case cbPre.Text
        Case "(último usado)": AplicarLista gUlt
        Case "(valores de fábrica)": AplicarLista gFab
        Case "(padrão)": AplicarTexto GetSetting("LTTools", Chave(), "_padrao", "")
        Case Else: AplicarTexto GetSetting("LTTools", Chave(), "P_" & cbPre.Text, "")
    End Select
End Sub

Private Sub AplicarLista(arr() As String)
    Dim i As Long
    For i = 1 To gN
        AplicarValor i, arr(i)
    Next
End Sub

Private Sub AplicarTexto(ByVal s As String)
    Dim p As Variant, i As Long
    If Len(s) = 0 Then Exit Sub
    p = Split(s, vbTab)
    For i = 1 To gN
        If i - 1 <= UBound(p) Then AplicarValor i, CStr(p(i - 1))
    Next
End Sub

Private Sub AplicarValor(ByVal i As Long, ByVal v As String)
    On Error Resume Next
    Select Case gTipo(i)
        Case 1, 4: ctl(i).Text = v
        Case 2: ctl(i).Value = (v = "1")
        Case 3
            If Val(v) >= 1 And Val(v) <= ctl(i).ListCount Then ctl(i).ListIndex = Val(v) - 1
    End Select
End Sub

Private Function ValorAtual(ByVal i As Long) As String
    Select Case gTipo(i)
        Case 1, 4: ValorAtual = Trim$(ctl(i).Text)
        Case 2: ValorAtual = IIf(ctl(i).Value, "1", "0")
        Case 3: ValorAtual = CStr(ctl(i).ListIndex + 1)
    End Select
End Function

Private Function LerValores() As String
    Dim i As Long, s As String
    For i = 1 To gN
        s = s & IIf(i > 1, vbTab, "") & ValorAtual(i)
    Next
    LerValores = s
End Function

    nome = Replace(nome, "|", "-")
    If Left$(nome, 1) = "(" Then nome = Mid$(nome, 2)
    SaveSetting "LTTools", Chave(), "P_" & nome, LerValores()
    lst = GetSetting("LTTools", Chave(), "_lista", "")
    If InStr("|" & lst & "|", "|" & nome & "|") = 0 Then
        lst = IIf(Len(lst) = 0, nome, lst & "|" & nome)
        SaveSetting "LTTools", Chave(), "_lista", lst
    End If
    PreencherLista nome
End Sub

Private Sub bExcluirPre_Click()
    Dim nome As String, v As Variant, novo As String
    If cbPre.ListIndex < 0 Then
        MsgBox "Escolha na lista a predefinição que quer excluir.", vbInformation, "LT Tools"
        Exit Sub
    End If
    nome = cbPre.Text
    If nome = "(padrão)" Then
        If MsgBox("Remover o padrão desta janela? (ela volta a abrir com os últimos valores usados)", vbQuestion + vbYesNo, "LT Tools") <> vbYes Then Exit Sub
        On Error Resume Next
        DeleteSetting "LTTools", Chave(), "_padrao"
        On Error GoTo 0
        PreencherLista ""
        Exit Sub
    End If
    If Left$(nome, 1) = "(" Then
        MsgBox "Esta opção não pode ser excluída.", vbInformation, "LT Tools"
        Exit Sub
    End If
    If MsgBox("Excluir a predefinição """ & nome & """?", vbQuestion + vbYesNo, "LT Tools") <> vbYes Then Exit Sub
    On Error Resume Next
    DeleteSetting "LTTools", Chave(), "P_" & nome
    On Error GoTo 0
    For Each v In Split(GetSetting("LTTools", Chave(), "_lista", ""), "|")
        If Len(v) > 0 And v <> nome Then novo = IIf(Len(novo) = 0, v, novo & "|" & v)
    Next
    SaveSetting "LTTools", Chave(), "_lista", novo
    PreencherLista ""
End Sub

Private Sub bPadraoPre_Click()
    SaveSetting "LTTools", Chave(), "_padrao", LerValores()
    PreencherLista "(padrão)"
    MsgBox "Pronto! Esta janela vai abrir sempre com estes valores.", vbInformation, "LT Tools"
End Sub
